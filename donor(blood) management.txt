import java.util.ArrayList;
import java.util.Scanner;

class Donor {
    int id;
    String name;
    String bloodGroup;
    String phone;

    Donor(int id, String name, String bloodGroup, String phone) {
        this.id = id;
        this.name = name;
        this.bloodGroup = bloodGroup;
        this.phone = phone;
    }

    void display() {
        System.out.println("\nDonor ID: " + id);
        System.out.println("Name: " + name);
        System.out.println("Blood Group: " + bloodGroup);
        System.out.println("Phone: " + phone);
    }
}

public class BloodDonorManagement {

    static Scanner sc = new Scanner(System.in);
    static ArrayList<Donor> donors = new ArrayList<>();
    static int id = 1;

    static void addDonor() {

        System.out.print("Enter donor name: ");
        String name = sc.nextLine();

        System.out.print("Enter blood group: ");
        String bloodGroup = sc.nextLine();

        System.out.print("Enter phone number: ");
        String phone = sc.nextLine();

        Donor donor = new Donor(id, name, bloodGroup, phone);
        donors.add(donor);

        System.out.println("\nDonor added successfully!");
        System.out.println("Donor ID: " + id);

        id++;
    }

    static void displayDonors() {

        if (donors.size() == 0) {
            System.out.println("\nNo donor records available.");
            return;
        }

        System.out.println("\n===== DONOR DETAILS =====");

        for (Donor donor : donors) {
            donor.display();
        }
    }

    static void searchByBloodGroup() {

        System.out.print("Enter blood group to search: ");
        String group = sc.nextLine();

        boolean found = false;

        System.out.println("\n===== AVAILABLE DONORS =====");

        for (Donor donor : donors) {

            if (donor.bloodGroup.equalsIgnoreCase(group)) {
                donor.display();
                found = true;
            }
        }

        if (!found) {
            System.out.println("No donor found for blood group " + group);
        }
    }

    static void removeDonor() {

        System.out.print("Enter donor ID to remove: ");
        int searchId = sc.nextInt();
        sc.nextLine();

        boolean found = false;

        for (Donor donor : donors) {

            if (donor.id == searchId) {
                donors.remove(donor);
                System.out.println("Donor removed successfully.");
                found = true;
                break;
            }
        }

        if (!found) {
            System.out.println("Donor not found.");
        }
    }

    public static void main(String[] args) {

        int choice;

        do {

            System.out.println("\n==============================");
            System.out.println("   BLOOD DONOR MANAGEMENT");
            System.out.println("==============================");
            System.out.println("1. Add Donor");
            System.out.println("2. View All Donors");
            System.out.println("3. Search by Blood Group");
            System.out.println("4. Remove Donor");
            System.out.println("5. Exit");

            System.out.print("Enter your choice: ");
            choice = sc.nextInt();
            sc.nextLine();

            switch (choice) {

                case 1:
                    addDonor();
                    break;

                case 2:
                    displayDonors();
                    break;

                case 3:
                    searchByBloodGroup();
                    break;

                case 4:
                    removeDonor();
                    break;

                case 5:
                    System.out.println("\nThank you! 🩸");
                    break;

                default:
                    System.out.println("\nInvalid choice.");
            }

        } while (choice != 5);

        sc.close();
    }
}
